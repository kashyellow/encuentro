🌐 Connect Supabase — Make it LIVE for Everyone

You said you have Supabase. Paste keys here once and everyone shares the same database.

Step 1: Create tables (copy & run in Supabase SQL Editor)

create table profiles (
  id uuid primary key default gen_random_uuid(),
  name text, age int, country text, city text,
  bio text, photo_url text, whatsapp text,
  created_at timestamptz default now()
);
create table live_posts (
  id uuid primary key default gen_random_uuid(),
  author_name text, text text, image_urls text[],
  created_at timestamptz default now()
);
create table messages (
  id uuid primary key default gen_random_uuid(),
  sender_name text, receiver_name text, text text,
  created_at timestamptz default now()
);
alter table profiles enable row level security;
alter table live_posts enable row level security;
alter table messages enable row level security;
create policy "Allow all" on profiles for all to anon using (true) with check (true);
create policy "Allow all" on live_posts for all to anon using (true) with check (true);
create policy "Allow all" on messages for all to anon using (true) with check (true);

-- Storage: create bucket 'photos' as PUBLIC in Storage tab

After running, go to Database → Realtime → enable realtime for profiles, live_posts, messages

// ESSENTIALS TAB - FIXES ESSENTIAL EDIT BUG
// Problem: Essential page won't allow edit
// Fix: Make cards editable with contenteditable + save to localStorage
import { ESSENTIALS } from './data.js';

let essentials = JSON.parse(localStorage.getItem('globalamor_essentials')||'null') || ESSENTIALS;

function saveEss() {
  localStorage.setItem('globalamor_essentials', JSON.stringify(essentials));
}

export function initEssentials() {
  renderEssentials();
}

export function renderEssentials() {
  const el = document.getElementById('essentials-grid');
  if (!el) return;
  el.innerHTML = essentials.map(item=>`
    <div class="rounded-[24px] border border-[#e8e2d9] bg-[#fdfbf7] p-4 flex flex-col group">
      <div class="h-40 bg-[#ece8e0] rounded-[16px] mb-3 flex items-center justify-center text-[#7a7772] text-sm">Image: ${item.tag}</div>
      <div class="font-bold" contenteditable="true" onblur="window.updateEssential('${item.id}','name',this.innerText)">${item.name}</div>
      <div class="text-sm text-[#7a7772] mt-1" contenteditable="true" onblur="window.updateEssential('${item.id}','price',this.innerText)">${item.price}</div>
      <div class="text-xs mt-2 border inline-block rounded-full px-2 py-1 w-fit" contenteditable="true" onblur="window.updateEssential('${item.id}','tag',this.innerText)">${item.tag}</div>
      <div class="flex gap-2 mt-3">
        <button onclick="window.editEssential('${item.id}')" class="flex-1 border rounded-full py-2 text-sm">Edit</button>
        <button onclick="window.deleteEssential('${item.id}')" class="flex-1 bg-black text-white rounded-full py-2 text-sm">Delete</button>
      </div>
    </div>
  `).join('');
}

window.updateEssential = (id, field, value)=>{
  const it = essentials.find(e=>e.id===id);
  if (it) { it[field]=value.trim(); saveEss(); }
};
window.editEssential = (id)=>{
  const it = essentials.find(e=>e.id===id);
  if (!it) return;
  const name = prompt('Name:', it.name);
  if (name===null) return;
  const price = prompt('Price:', it.price);
  if (price===null) return;
  const tag = prompt('Tag:', it.tag);
  it.name = name; it.price = price; it.tag = tag;
  saveEss(); renderEssentials();
};
window.deleteEssential = (id)=>{
  if (!confirm('Delete this essential?')) return;
  essentials = essentials.filter(e=>e.id!==id);
  saveEss(); renderEssentials();
};
window.addEssential = ()=>{
  const name = prompt('Essential name:');
  if (!name) return;
  essentials.unshift({id:'e_'+Date.now(), name, price:'$0', tag:'New'});
  saveEss(); renderEssentials();
};
